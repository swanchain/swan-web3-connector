Usage Example

// Configure supported wallets WalletButton.configureSupportedWallets([ { id: 'metamask', name: 'MetaMask', icon: 'https://raw.githubusercontent.com/MetaMask/brand-resources/master/SVG/metamask-fox.svg', getProvider: () => window.ethereum }, { id: 'coinbase', name: 'Coinbase Wallet', icon: 'https://www.coinbase.com/assets/press/coinbase-icon.png', getProvider: () => window.ethereum } ]); // Default button const walletButton = new WalletButton({ buttonText: 'Connect Wallet', onConnect: (wallet) => { console.log('Connected:', wallet.account); }, onDisconnect: () => { console.log('Disconnected'); } }); // Dark theme button const darkThemeButton = new WalletButton({ buttonText: 'Connect Wallet', styles: { button: { background: '#1a1a1a', color: '#ffffff', border: 'none', padding: '12px 24px', borderRadius: '12px', fontSize: '15px', fontWeight: '500', boxShadow: '0 2px 8px rgba(0, 0, 0, 0.2)', hoverBackground: '#2a2a2a', hoverBorderColor: 'transparent', hoverBoxShadow: '0 4px 12px rgba(0, 0, 0, 0.3)' }, connected: { background: '#2a2a2a', borderColor: 'transparent', hoverBackground: '#333333' }, balance: { color: '#4CAF50', fontWeight: '600' }, address: { color: '#ffffff', fontFamily: "'SFMono-Regular', Consolas, 'Liberation Mono', Menlo, monospace" } } }); // Minimal style button const minimalButton = new WalletButton({ buttonText: 'Connect Wallet', styles: { button: { background: 'transparent', color: '#2196F3', border: '1px solid #2196F3', padding: '10px 20px', borderRadius: '8px', fontSize: '14px', fontWeight: '500', boxShadow: 'none', hoverBackground: '#f5f5f5', hoverBorderColor: '#1976D2', hoverBoxShadow: 'none' }, connected: { background: '#f5f5f5', borderColor: '#e0e0e0', hoverBackground: '#eeeeee' }, balance: { color: '#2196F3', fontWeight: '500' }, address: { color: '#666666', fontFamily: "'SFMono-Regular', Consolas, 'Liberation Mono', Menlo, monospace" } } }); // Gradient style button const gradientButton = new WalletButton({ buttonText: 'Connect Wallet', styles: { button: { background: 'linear-gradient(45deg, #2196F3, #1976D2)', color: '#ffffff', border: 'none', padding: '12px 24px', borderRadius: '12px', fontSize: '15px', fontWeight: '500', boxShadow: '0 2px 8px rgba(33, 150, 243, 0.3)', hoverBackground: 'linear-gradient(45deg, #1976D2, #1565C0)', hoverBorderColor: 'transparent', hoverBoxShadow: '0 4px 12px rgba(33, 150, 243, 0.4)' }, connected: { background: 'linear-gradient(45deg, #4CAF50, #388E3C)', borderColor: 'transparent', hoverBackground: 'linear-gradient(45deg, #388E3C, #2E7D32)' }, balance: { color: '#ffffff', fontWeight: '600' }, address: { color: '#ffffff', fontFamily: "'SFMono-Regular', Consolas, 'Liberation Mono', Menlo, monospace" } } }); walletButton.mount('#wallet-container'); darkThemeButton.mount('#wallet-container-dark'); minimalButton.mount('#wallet-container-minimal'); gradientButton.mount('#wallet-container-gradient');
